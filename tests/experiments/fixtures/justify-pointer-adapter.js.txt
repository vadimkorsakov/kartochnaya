// Diagnostic-only code inside the existing session closure: one cancellable post-click task.
// No independent text positions, document flags or replacement Selection model.
let pointerEpoch=0,pointerGesture=null;
const pointerEmit=(type,event,extra={})=>{
 let selection=null;try{if(selectionInside())selection=readSelection()}catch{}
 if(extra.hit){const {key,...publicHit}=extra.hit;extra={...extra,hit:publicHit}}
 onPointerDiagnostic({type,epoch:pointerEpoch,event:event?.type,detail:event?.detail,x:event?.clientX,y:event?.clientY,shift:event?.shiftKey,selection,...extra});
};
const clearPointerGesture=(reason,event)=>{const p=pointerGesture;pointerGesture=null;if(p?.task!==undefined)diagnosticCancelPointerTask(p.task);if(p)pointerEmit('cancel',event,{reason});};
if(diagnosticPointerAdapter){
 const doc=host.ownerDocument;
 const watch=(target,type,handler,capture=false)=>target.addEventListener(type,handler,{capture,signal:abort.signal});
 function stretchedHit(clientX,clientY){
  if(layer.textAlign!=='justify'||inactive||fallback||composing||safe.phase!=='confirmed'||!clean())return null;
  const rect=host.getBoundingClientRect(),css=getComputedStyle(host),matrix=new DOMMatrix(css.transform==='none'?undefined:css.transform);
  if(matrix.b||matrix.c||matrix.a<=0||matrix.d<=0)return null;
  const sx=rect.width/parseFloat(css.width),sy=rect.height/parseFloat(css.height);
  if(!(sx>0&&sy>0))return null;
  const x=(clientX-rect.left)/sx,y=(clientY-rect.top)/sy,projection=layout(layer);
  const key=JSON.stringify([layer,[rect.x,rect.y,rect.width,rect.height],projection.rows.map(r=>[r.start,r.end,r.y,r.height,r.carets,r.justification])]);
  for(const [rowIndex,row] of projection.rows.entries()){
   if(y<row.y||y>=row.y+row.height||!(row.justification?.extraPerSpace>0))continue;
   for(const start of row.justification.spaces){
    if(layer.text[start]!==' ')continue;
    const before=row.carets.find(p=>p.offset===start),after=row.carets.find(p=>p.offset===start+1);
    if(!before||!after||!(after.x>before.x)||x<before.x||x>after.x)continue;
    return {rowIndex,start,end:start+1,target:x<=(before.x+after.x)/2?start:start+1,key,x,before:before.x,after:after.x};
   }
  }
  return null;
 }
 listen('pointerdown',event=>{
  clearPointerGesture('new-pointer',event);
  if(!event.isTrusted||event.pointerType!=='mouse'||!event.isPrimary||event.button!==0||event.altKey||event.ctrlKey||event.metaKey)return;
  const hit=stretchedHit(event.clientX,event.clientY);if(!hit)return;
  let anchor=null,anchorNode=null,anchorOffset=null;
  if(event.shiftKey){try{if(document.activeElement!==host||!selectionInside())return;anchor=readSelection().anchor;anchorNode=getSelection().anchorNode;anchorOffset=getSelection().anchorOffset}catch{return}}
  pointerGesture={id:event.pointerId,x:event.clientX,y:event.clientY,epoch:pointerEpoch,key:hit.key,start:hit.start,rowIndex:hit.rowIndex,anchor,anchorNode,anchorOffset,shift:event.shiftKey,up:false};
  pointerEmit('candidate',event,{hit});
 });
 watch(doc,'pointermove',event=>{const p=pointerGesture;if(p&&!p.clicked&&p.id===event.pointerId&&(event.clientX!==p.x||event.clientY!==p.y))clearPointerGesture('movement',event)},true);
 watch(doc,'pointerup',event=>{
  const p=pointerGesture;if(!p||p.id!==event.pointerId)return;
  if(!host.contains(event.target)||event.clientX!==p.x||event.clientY!==p.y){clearPointerGesture('pointerup-outside-or-moved',event);return}
  p.up=true;pointerEmit('pointerup-observed',event);
 },true);
 watch(doc,'pointercancel',event=>clearPointerGesture('pointercancel',event),true);
 listen('dragstart',event=>clearPointerGesture('native-dragstart',event));
 listen('focusout',event=>clearPointerGesture('focusout',event));
 watch(window,'blur',event=>clearPointerGesture('window-blur',event));
 listen('compositionstart',event=>clearPointerGesture('composition',event));
 function flushBeforeInput(event){
  if(event.isComposing||event.keyCode===229||composing){clearPointerGesture('composing-input',event);return}
  if(pointerGesture?.clicked)applyClick(event);else clearPointerGesture('input-before-click',event);
 }
 listen('keydown',flushBeforeInput);
 listen('beforeinput',flushBeforeInput);
 watch(doc,'selectionchange',event=>{if(event.isTrusted&&pointerGesture?.clicked)applyClick(event)});
 function applyClick(event){
  const p=pointerGesture;if(!p?.clicked)return;
  pointerGesture=null;if(p.task!==undefined)diagnosticCancelPointerTask(p.task);
  const hit=stretchedHit(p.cx,p.cy);
  if(document.activeElement!==host||!hit||p.epoch!==pointerEpoch||p.key!==hit.key||p.start!==hit.start||p.rowIndex!==hit.rowIndex){pointerEmit('ignored',event,{reason:'stale-or-ineligible'});return}
  let native;try{native=readSelection()}catch{pointerEmit('ignored',event,{reason:'outside-selection'});return}
  if(!p.shift&&native.anchor!==native.focus){pointerEmit('ignored',event,{reason:'native-range'});return}
  if(p.shift&&!host.contains(p.anchorNode)){pointerEmit('ignored',event,{reason:'stale-native-anchor'});return}
  const anchor=p.shift?p.anchor:hit.target;
  const sameAnchor=!p.shift||(getSelection().anchorNode===p.anchorNode&&getSelection().anchorOffset===p.anchorOffset);
  if(native.anchor===anchor&&native.focus===hit.target&&sameAnchor){pointerEmit('already-correct',event,{hit});return}
  setSelection(anchor,hit.target,hit.rowIndex);
  if(p.shift){const s=getSelection();s.setBaseAndExtent(p.anchorNode,p.anchorOffset,s.focusNode,s.focusOffset)}
  pointerEmit('correct',event,{hit,anchor,focus:hit.target,clickX:p.cx,clickY:p.cy});
 }
 listen('click',event=>{
  const p=pointerGesture;
  if(!p)return;
  if(!p.up||!event.isTrusted||event.button!==0||event.detail!==1||event.altKey||event.ctrlKey||event.metaKey||event.shiftKey!==p.shift||document.activeElement!==host){clearPointerGesture('invalid-click',event);return}
  const hit=stretchedHit(event.clientX,event.clientY);
  if(!hit||p.epoch!==pointerEpoch||p.key!==hit.key||p.start!==hit.start||p.rowIndex!==hit.rowIndex){clearPointerGesture('stale-click',event);return}
  p.clicked=true;p.cx=event.clientX;p.cy=event.clientY;
  pointerEmit('pending-click',event,{hit});
  // A native Selectionchange may run first. Input guards consume the same
  // token if the browser prioritizes keyboard/beforeinput over the timer task.
  p.task=diagnosticSchedulePointerTask(()=>{if(pointerGesture===p)applyClick({type:'post-click-task'})});
 });
}
